# Setup at home

/dev/ttyACM0

lerobot-setup-motors \
    --robot.type=so101_follower \
    --robot.port=/dev/ttyACM0
    
lerobot-calibrate \
    --robot.type=so101_follower \
    --robot.port=/dev/ttyACM0 \
    --robot.id=so101_follower_arm
    
/dev/ttyACM1

lerobot-setup-motors \
    --robot.type=so101_leader\
    --robot.port=/dev/ttyACM1


lerobot-calibrate \
    --teleop.type=so101_leader \
    --teleop.port=/dev/ttyACM1
    --teleop.id=so101_leader_arm

lerobot-calibrate \
    --robot.type=so101_follower \
    --robot.port=/dev/ttyACM1 \
    --robot.id=so101_leader_arm
    

lerobot-teleoperate \
    --robot.type=so101_follower \
    --robot.port=/dev/ttyACM0 \
    --robot.id=so101_follower_arm \
    --robot.cameras="{front: {type: opencv, index_or_path: /dev/video2, width: 640, height: 480, fps: 30}, wrist: {type: opencv, index_or_path: /dev/video4, width: 640, height: 480, fps: 30}}" \
    --teleop.type=so101_leader \
    --teleop.port=/dev/ttyACM1 \
    --teleop.id=so101_leader_arm \
    --display_data=true
    

lerobot-teleoperate \
    --robot.type=so101_follower \
    --robot.port=/dev/ttyACM0 \
    --robot.id=so101_follower_arm \
    --robot.cameras="{front: {type: opencv, index_or_path: /dev/video2, backend: V4L2, width: 640, height: 480, fps: 15}, wrist: {type: opencv, index_or_path: /dev/video4, backend: V4L2,  width: 640, height: 480, fps: 15}}" \
    --teleop.type=so101_leader \
    --teleop.port=/dev/ttyACM1 \
    --teleop.id=so101_leader_arm \
    --display_data=true
